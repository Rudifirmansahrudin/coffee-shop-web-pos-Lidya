type Json = Record<string, unknown>;

const priceFormatter = new Intl.NumberFormat("id-ID", {
  style: "currency",
  currency: "IDR",
  maximumFractionDigits: 0,
});

function formatPrice(value: number) {
  // Intl gives "Rp 25.000" with a non-breaking space; normalize it
  return priceFormatter.format(value).replace(/\s/g, " ");
}

function asObject(value: unknown): Json {
  return value && typeof value === "object" && !Array.isArray(value) ? (value as Json) : {};
}

function asString(value: unknown): string | null {
  return typeof value === "string" && value.trim() ? value.trim() : null;
}

function buildWhatsAppUrl(raw: string | null) {
  if (!raw) return null;
  if (raw.startsWith("http")) return raw;
  let digits = raw.replace(/\D/g, "");
  if (digits.startsWith("0")) digits = `62${digits.slice(1)}`;
  return digits ? `https://wa.me/${digits}` : null;
}

function buildInstagramUrl(raw: string | null) {
  if (!raw) return null;
  if (raw.startsWith("http")) return raw;
  return `https://instagram.com/${raw.replace(/^@/, "")}`;
}

import Image from "next/image";
import type { CSSProperties } from "react";
import { cn } from "@/lib/utils";

const ORNAMENTS = {
  astronaut: { src: "/ornaments/astronaut.png", w: 370, h: 270 },
  rocket: { src: "/ornaments/rocket.png", w: 120, h: 180 },
  comet: { src: "/ornaments/comet.png", w: 240, h: 215 },
  planet: { src: "/ornaments/planet-ring.png", w: 258, h: 155 },
  "ufo-dome": { src: "/ornaments/ufo-dome.png", w: 205, h: 175 },
  "ufo-ship": { src: "/ornaments/ufo-ship.png", w: 190, h: 120 },
} as const;

export type OrnamentName = keyof typeof ORNAMENTS;

type Props = {
  name: OrnamentName;
  /** Rendered width in px (height follows the image ratio) */
  width?: number;
  /** Position it with className, e.g. "right-4 top-6". Parent must be `relative`. */
  className?: string;
  /** Slow floating motion, dark mode only */
  float?: boolean;
  /** Float speed variant: 1 = 7s, 2 = 9s, 3 = 11s (desync several ornaments) */
  speed?: 1 | 2 | 3;
  /** Base tilt in degrees */
  rotate?: number;
};

// Decorative only: light mode = faint silhouette watermark, dark mode = full art
export function Ornament({
  name,
  width = 120,
  className,
  float = true,
  speed = 1,
  rotate = 0,
}: Props) {
  const item = ORNAMENTS[name];

  return (
    <Image
      src={item.src}
      alt=""
      aria-hidden="true"
      width={item.w}
      height={item.h}
      sizes={`${width}px`}
      style={{ width, height: "auto", "--r": `${rotate}deg` } as CSSProperties}
      className={cn(
        "pointer-events-none absolute select-none brightness-0 opacity-10 dark:brightness-100 dark:opacity-90",
        float && "orn-float",
        float && speed === 2 && "orn-s2",
        float && speed === 3 && "orn-s3",
        className
      )}
    />
  );
}
